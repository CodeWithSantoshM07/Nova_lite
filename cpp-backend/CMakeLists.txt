cmake_minimum_required(VERSION 3.16)
project(bank CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(CROW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CROW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CROW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(Crow GIT_REPOSITORY https://github.com/CrowCpp/Crow.git GIT_TAG v1.2.0)
FetchContent_MakeAvailable(Crow)
# MySQL Connector/C++ (legacy JDBC API). On Windows pass -DMYSQLCPP_DIR="C:/Program Files/MySQL/MySQL Connector C++ 8.x"
find_path(MYSQLCPP_INC mysql_driver.h HINTS ${MYSQLCPP_DIR}/include/jdbc PATH_SUFFIXES jdbc mysql-cppconn-8/jdbc mysql-cppconn-8)
find_library(MYSQLCPP_LIB NAMES mysqlcppconn HINTS ${MYSQLCPP_DIR}/lib64 ${MYSQLCPP_DIR}/lib64/vs14)
add_executable(bank main.cpp)
target_include_directories(bank PRIVATE ${MYSQLCPP_INC})
target_compile_definitions(bank PRIVATE FRONTEND="${CMAKE_CURRENT_SOURCE_DIR}/../frontend")
target_link_libraries(bank PRIVATE Crow::Crow ${MYSQLCPP_LIB})
